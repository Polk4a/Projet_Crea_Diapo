\begin{frame}{Ce qu'on affirme, ce qu'on ne promet pas}
  \begin{columns}[T]
    \begin{column}{0.485\textwidth}
      \begin{encadre}{À mettre en avant}
        \footnotesize\raggedright
        \oui\textbf{Payé une fois}~: 69\,€, sans abonnement (voir la partie Concurrence).\par\vspace{1pt}
        \oui\textbf{Marche sans opérateur}~: des messages entre voisins quand le réseau tombe.\par\vspace{1pt}
        \oui\textbf{Prêt à l'emploi, en français}~: réglages Europe, clé privée, guide.\par\vspace{1pt}
        \oui\textbf{Libre et vérifiable}~: code ouvert, ni compte ni serveur central.\par\vspace{1pt}
        \oui\textbf{Compatible Meshtastic}~: avec les logos du projet, jamais dans notre nom.\par
      \end{encadre}
    \end{column}
    \begin{column}{0.485\textwidth}
      \begin{remarque}[À ne jamais promettre]
        \footnotesize\raggedright
        \non\textbf{Un secours}~: ce n'est pas une balise de détresse, aucun centre de secours n'est relié.\par\vspace{1pt}
        \non\textbf{Une portée garantie}~: elle dépend du relief, de l'antenne et des voisins.\par\vspace{1pt}
        \non\textbf{Un téléphone}~: ni appels ni internet, du texte seulement.\par\vspace{1pt}
        \non\textbf{Un réseau déjà là}~: il faut des voisins équipés pour que le kit serve.\par\vspace{1pt}
        \non\textbf{Un produit «\,Meshtastic\,»}~: la marque appartient au projet libre.\par
      \end{remarque}
    \end{column}
  \end{columns}
  \vspace{0.08cm}
  \begin{pcompact}\centering\footnotesize
    Une promesse étroite et tenue protège la marque et réduit le risque juridique.
  \end{pcompact}
  \srcnote{Règles d'usage de la marque~: \cite{meshtastic_licence}. Risque juridique~: frein F3 de l'étude documentaire. Portées indicatives~: partie Fonctionnement.}
\end{frame}
