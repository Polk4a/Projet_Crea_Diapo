% --- barres horizontales à échelle logarithmique -----------------------------
% réglages : \mkox = début des barres (cm), \mkdec = cm par décade, \mkmin = log10 de la valeur à l'origine,
%            \mklabw = largeur du libellé à gauche
\newcommand{\mkox}{3.0}\newcommand{\mkdec}{1.0}\newcommand{\mkmin}{3}\newcommand{\mklabw}{2.8cm}
% \mkbl{y}{mantisse}{exposant}{libellé}{texte affiché}{couleur}  : valeur = mantisse x 10^exposant
% (pgfmath ne lit pas un entier supérieur à 16 383, d'où cette écriture)
\newcommand{\mkbl}[6]{%
  \pgfmathsetmacro{\mkL}{\mkdec*(log10(#2)+#3-\mkmin)}%
  \fill[#6] (\mkox,#1-0.15) rectangle ++(\mkL,0.30);%
  \node[anchor=east,text width=\mklabw,align=right,font=\scriptsize,inner sep=1pt] at (\mkox-0.08,#1) {#4};%
  \node[anchor=west,font=\scriptsize\bfseries,inner sep=1pt] at ({\mkox+\mkL+0.06},#1) {#5};}
% axe des décades sous les barres : \mkaxe{y}{première décade}{dernière décade}
\newcommand{\mkaxe}[3]{%
  \draw[insagris,thin] (\mkox,#1) -- ({\mkox+\mkdec*(#3-\mkmin)},#1);%
  \foreach \mkd in {#2,...,#3} {%
    \draw[insagris,thin] ({\mkox+\mkdec*(\mkd-\mkmin)},#1) -- ++(0,-0.07);%
    \node[anchor=north,font=\tiny,text=insagris,inner sep=1pt] at ({\mkox+\mkdec*(\mkd-\mkmin)},#1-0.06) {$10^{\mkd}$};}}
